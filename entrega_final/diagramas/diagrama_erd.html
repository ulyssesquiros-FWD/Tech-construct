Diccionario de Estructura de Datos (14 Tablas Normalizadas)
roles Seguridad
  • PK id INT AUTO
  • name VARCHAR(50) UK
  • description VARCHAR(255)
users Seguridad
  • PK id INT AUTO
  • FK role_id INT
  • full_name VARCHAR(100)
  • email VARCHAR(100) UK
  • password_hash VARCHAR(255)
  • department VARCHAR(100)
  • status ENUM(activo,inactivo)
categories Catálogo
  • PK id INT AUTO
  • code VARCHAR(20) UK
  • name VARCHAR(100)
  • FK parent_id INT NULL
materials Catálogo
  • PK id INT AUTO
  • internal_code VARCHAR(50) UK
  • name VARCHAR(150)
  • FK category_id INT
  • FK subcategory_id INT
  • unit VARCHAR(20)
  • avg_daily_demand DECIMAL(10,2)
  • max_daily_demand DECIMAL(10,2)
  • lead_time_days INT
  • safety_stock DECIMAL(10,2)
  • reorder_point DECIMAL(10,2)
  • standard_price DECIMAL(12,2)
locations Logística
  • PK id INT AUTO
  • code VARCHAR(20) UK
  • name VARCHAR(100)
  • type ENUM(bodega,obra)
  • address TEXT
  • status ENUM
suppliers Compras
  • PK id INT AUTO
  • code VARCHAR(20) UK
  • commercial_name VARCHAR(150)
  • tax_id VARCHAR(50) UK
  • phone / email VARCHAR
  • rating DECIMAL(3,2)
purchase_orders Compras
  • PK id INT AUTO
  • po_number VARCHAR(30) UK
  • FK supplier_id INT
  • FK location_id INT
  • FK created_by INT
  • order_date DATE
  • subtotal / tax / total DECIMAL
  • status ENUM
  • quotations_count INT
  • required_approver_role VARCHAR
purchase_order_details Compras
  • PK id INT AUTO
  • FK purchase_order_id INT
  • FK material_id INT
  • quantity DECIMAL(10,2)
  • unit_price DECIMAL(12,2)
  • subtotal DECIMAL(12,2)
  • received_quantity DECIMAL(10,2)
approvals Compras
  • PK id INT AUTO
  • FK purchase_order_id INT
  • FK user_id INT
  • decision ENUM(aprobada,rechazada)
  • comments TEXT
  • decided_at DATETIME
inventory_movements Libro Mayor (Ledger)
  • PK id INT AUTO
  • movement_type ENUM(entrada,salida,traslado,ajuste)
  • FK material_id INT
  • FK origin_location_id INT NULL
  • FK destination_location_id INT NULL
  • quantity DECIMAL(10,2)
  • unit_cost DECIMAL(12,2)
  • reference_document VARCHAR(100)

  • PK id INT AUTO
  • FK material_id INT

  • urgency ENUM(baja,media,alta,critica)
